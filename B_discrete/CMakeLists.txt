cmake_minimum_required(VERSION 3.15)
project(B_discrete LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(ortools QUIET CONFIG PATHS "$ENV{HOME}/.local/ortools")

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
    add_executable(B_discrete_main src/main.cpp)
    target_include_directories(B_discrete_main PRIVATE src)
    if(ortools_FOUND)
        target_link_libraries(B_discrete_main PRIVATE ortools::ortools)
        target_compile_definitions(B_discrete_main PRIVATE HAS_ORTOOLS=1)
    endif()
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_main.cpp")
    add_executable(B_discrete_test tests/test_main.cpp)
    target_include_directories(B_discrete_test PRIVATE src)
    if(ortools_FOUND)
        target_link_libraries(B_discrete_test PRIVATE ortools::ortools)
        target_compile_definitions(B_discrete_test PRIVATE HAS_ORTOOLS=1)
    endif()
endif()
